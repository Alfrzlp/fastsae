fastsae

Fast Small Area Estimation

High-performance R package for Small Area Estimation using Fay-Herriot, Spatial Fay-Herriot, and Battese-Harter-Fuller models powered by C++.

up to 12,000x faster RcppArmadillo Parallel bootstrap GPL-3.0 license

Installation

# Github Version
install.packages("remotes")
remotes::install_github("ridsonap/fastsae")

# CRAN Version
install.packages("fastsae")

Supported Models

Function Model MSE
eblup_area() Fay-Herriot (area-level) Analytical
seblup_area() Spatial Fay-Herriot Analytical, PBMSE, NPBMSE
eblup_unit() Battese-Harter-Fuller (unit-level) PBMSE

eblup_area — Fay-Herriot

library(fastsae)

m1 <- eblup_area(
  y ~ x1 + x2 + x3,
  data = mys,
  vardir = "vardir",
  method = "REML"
)

Output: df_eblup, estcoef, random_effect_var, goodness, n_iter, convergence

seblup_area — Spatial Fay-Herriot

m2 <- seblup_area(
  y ~ x1 + x2 + x3,
  data = mys,
  vardir = "vardir",
  W = mys_proxmat,
  method = "REML"
)

Parametric Bootstrap MSE:

m3 <- seblup_area(
  y ~ x1 + x2 + x3,
  data = mys,
  vardir = "vardir",
  method = "REML",
  W = mys_proxmat,
  mse_method = "pbmse",
  B = 50,
  n_threads = 4,
  seed = 1
)

eblup_unit — Battese-Harter-Fuller

library(dplyr)

# Prepare data
df_meanpop <- cornsoybeanmeans |>
  rename(CornPix = MeanCornPixPerSeg, SoyBeansPix = MeanSoyBeansPixPerSeg)
df_cornsoybean <- cornsoybean |>
  rename(CountyIndex = County)

m3 <- eblup_unit(
  formula = CornHec ~ CornPix + SoyBeansPix,
  unit_data = df_cornsoybean,
  Xpop = df_meanpop,
  domain_var = "CountyIndex",
  popsize_var = "PopnSegments"
)
